# build image 
# For Nginx + Gunicorn + Uvicorn
# Dockerfile=Dockerfile.nginx
docker build -t fastapi-nginx .

# start container
docker run -d -p 8080:80 --name fastapi-test fastapi-nginx

# Dockerfile without Nginx
docker build -t fastapi-test-image .

docker run -d -p 8080:8000 --name fastapi-test fastapi-test-image

# 3 check running containers
docker ps

# stop container
docker stop fastapi-test

# remove container
docker rm fastapi-test

# check container log
docker logs fastapi-test

# try interactive mode:

docker run -it fastapi-nginx /bin/bash

# Then manually test:

nginx -t
gunicorn n2t.app:app -k uvicorn.workers.UvicornWorker --bind 127.0.0.1:8000
