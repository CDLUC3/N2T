# build image 
# For Gunicorn + Uvicorn
docker build -t fastapi-nginx -f Dockerfile.local .

# start container
docker run -d -p 8000:18880 --name fastapi-test fastapi-nginx

# run multiple containers on different host ports
docker run -d -p 80:18880 --name container_1 fastapi-nginx
docker run -d -p 8080:18880 --name container_2 fastapi-nginx
docker run -d -p 18880:18880 --name container_3 fastapi-nginx


# build a docker image that only contains nginx
docker build -t nginx-web -f Dockerfile.nginx .

docker run -d -p 8080:8080 --name nginx-web-1 nginx-web

# 3 check running containers
docker ps

# stop container
docker stop fastapi-test

# remove container
docker rm fastapi-test

# check container log
docker logs fastapi-test

# view the content of a container
docker exec -it fastapi-test /bin/bash

# try interactive mode:

docker run -it fastapi-nginx /bin/bash

# Then manually test:

nginx -t
gunicorn n2t.app:app -k uvicorn.workers.UvicornWorker --bind 127.0.0.1:8000
